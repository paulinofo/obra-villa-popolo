function fusionarDoc(docs, d, doc){
  if(!doc||typeof doc!=='object'||!doc.id||typeof doc.t!=='number') return false;
  const col=docs[String(d)]||(docs[String(d)]={});
  const act=col[doc.id];
  if(act&&act.t>=doc.t) return false;
  col[doc.id]=doc; return true;
}
function fusionarCeldas(docs, celdas){
  let n=0;
  (celdas||[]).forEach(c=>{
    if(!c||typeof c.d!=='number'||c.d<0||c.d>=COLECCIONES.length) return;
    if(fusionarDoc(docs,c.d,c.v)) n++;
  });
  return n;
}

function urlSala(sala){ return SYNC_URL+'/api/torneo/'+sala+'/scores'; }
async function sync(){
  if(!S||!S.sala){ estadoSync.estado='sin clave'; onSync(); return false; }
  if(syncOcupado) return false;
  syncOcupado=true;
  let hubo=false;
  try{
    const pendientes=S.cola.slice();
    const opts={ method:pendientes.length?'POST':'GET', headers:{'X-Torneo-Token':SYNC_TOKEN,'Content-Type':'application/json'} };
    if(pendientes.length) opts.body=JSON.stringify({celdas:pendientes});
    const r=await fetch(urlSala(S.sala), opts);
    if(r.status===401) throw new Error('token');
    if(!r.ok) throw new Error('http '+r.status);
    /* la fusión gana por sello del teléfono que escribe: un reloj desajustado
       haría que sus capturas pierdan contra datos viejos. Lo medimos contra la
       hora del servidor para poder avisarlo. */
    try{
      const hd=r.headers&&r.headers.get&&r.headers.get('Date');
      if(hd){ const srv=Date.parse(hd); if(srv) estadoSync.desfase=srv-Date.now(); }
    }catch(e){}
    const j=await r.json();
    const celdas=(j&&j.celdas)||[];
    hubo=fusionarCeldas(S.docs, celdas)>0;
    const eco={}; celdas.forEach(c=>{ const k=c.d+'.'+c.jid; eco[k]=Math.max(eco[k]||0, c.t||0); });
    S.cola=S.cola.filter(c=>!((eco[c.d+'.'+c.jid]||0)>=c.t));
    estadoSync.estado='en vivo'; estadoSync.ultimo=Date.now(); S.ultimaSync=estadoSync.ultimo;
    guardar();
    if(hubo) onCambio();
  }catch(e){
    estadoSync.estado= e.message==='token'?'token':'sin señal';
  }
  syncOcupado=false; onSync();
  return hubo;
}
function programarSync(ms){
  if(typeof module!=='undefined') return;   // en pruebas se llama sync() a mano
  clearTimeout(syncTimer); syncTimer=setTimeout(sync, ms);
}

function relojDesfasado(){ return Math.abs(estadoSync.desfase)>120000; }
function textoReloj(){
  if(!estadoSync.ultimo) return 'sin comprobar';
  if(!relojDesfasado()) return 'al día';
  const min=Math.round(Math.abs(estadoSync.desfase)/60000);
  return 'este teléfono va '+(estadoSync.desfase>0?'atrasado':'adelantado')+' '+min+' min · corrige la hora en Ajustes de iOS para que tus capturas no se pierdan';
}

function guardarRubro(id){
  const f=leerForm(['nombre','cant','moU','matU','nota']); const base=S.docs['1'][id];
  const cant=numero(f.cant), moU=numero(f.moU), matU=numero(f.matU);
  if(!f.nombre.trim()||isNaN(cant)||cant<0||isNaN(moU)||isNaN(matU)){ alert('Revisa nombre, cantidad y precios.'); return; }
  escribir(1,Object.assign({},base,{ nombre:f.nombre.trim(), cant, moU, matU, nota:f.nota.trim() })); abrirRubro(id);
}
